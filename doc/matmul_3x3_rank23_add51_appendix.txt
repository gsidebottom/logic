/-- The 51-addition-per-step scheme as a map on `3x3` matrices, in the
structure of `Matmul55.scheme`: the two UNCOUNTED unimodular input
basis changes, the COUNTED 12- and 13-addition input chains, the 23
products, the COUNTED 26-addition transposed output stage, and the
UNCOUNTED output basis change.  Counted additions per recursive step:
12 + 13 + 26 = **51**. -/
def rank23_add51_scheme (A B : Matrix (Fin 3) (Fin 3) R) :
    Matrix (Fin 3) (Fin 3) R :=
  -- A-side basis change (UNCOUNTED, amortized): the 9 transformed
  -- inputs, an invertible integer map applied once per matrix
  let ab0 := A 0 0
  let ab1 := A 1 0
  let ab2 := A 0 0 + A 2 0
  let ab3 := A 0 1 + A 2 1
  let ab4 := A 1 1
  let ab5 := A 2 1
  let ab6 := A 1 1 + A 1 2 - A 2 2
  let ab7 := A 2 2
  let ab8 := A 0 2
  -- A-side chain: 12 COUNTED additions (paid per recursive step)
  let aw0 := ab0 + ab5
  let aw1 := ab1 - ab7
  let aw2 := ab2 - ab7
  let aw3 := ab3 + ab8
  let aw4 := ab4 - ab8
  let aw5 := ab6 - aw1
  let aw6 := aw0 - aw3
  let aw7 := aw2 - aw3
  let aw8 := aw4 - aw5
  let aw9 := aw4 - aw6
  let aw10 := ab1 - ab2
  let aw11 := aw0 + aw10
  -- B-side basis change (UNCOUNTED)
  let bb0 := B 0 0
  let bb1 := B 0 1 + B 1 1
  let bb2 := B 0 2 + B 1 2
  let bb3 := B 1 1
  let bb4 := B 1 2
  let bb5 := B 2 0 - B 2 2
  let bb6 := B 1 1 - B 2 1
  let bb7 := B 2 2
  let bb8 := B 1 0
  -- B-side chain: 13 COUNTED additions
  let bw0 := bb0 - bb2
  let bw1 := bb0 + bb7
  let bw2 := bb4 + bw0
  let bw3 := bb5 + bw1
  let bw4 := bb6 + bb7
  let bw5 := bb4 - bw4
  let bw6 := bb5 + bw4
  let bw7 := bb3 - bw6
  let bw8 := bb8 + bw0
  let bw9 := bb8 - bw6
  let bw10 := bw1 - bw2
  let bw11 := bb0 - bb1
  let bw12 := bw6 + bw11
  -- the 23 products (A factor on the left: recurses on blocks)
  let m1 := aw0 * bw0
  let m2 := aw1 * bw1
  let m3 := aw2 * bb0
  let m4 := ab0 * bb1
  let m5 := ab1 * bw12
  let m6 := ab2 * bw12
  let m7 := ab0 * bb2
  let m8 := ab1 * bw10
  let m9 := aw11 * bw2
  let m10 := ab3 * bw9
  let m11 := ab4 * bw9
  let m12 := ab5 * bw8
  let m13 := aw6 * bb3
  let m14 := aw4 * bw4
  let m15 := aw3 * bw6
  let m16 := aw9 * bb4
  let m17 := ab4 * bw5
  let m18 := ab6 * bb5
  let m19 := ab7 * bw3
  let m20 := ab8 * bb6
  let m21 := aw5 * bw7
  let m22 := aw7 * bw7
  let m23 := aw8 * bb7
  -- output stage: 26 COUNTED additions -- the Tellegen transpose
  -- of the certified 12-addition C chain (12 + (23-9) fan-in = 26)
  let z0 := m3 + m19
  let z1 := m14 + m20
  let z2 := z1 + m17
  let z3 := m23 - z2
  let z4 := m2 + m19
  let z5 := z4 - z3
  let z6 := m1 - z0
  let z7 := z6 + m7
  let z8 := m15 - m20
  let z9 := z8 + z0
  let z10 := m18 - m17
  let z11 := z10 + z5
  let z12 := m4 - m13
  let z13 := z12 - m20
  let z14 := m5 - m21
  let z15 := z14 - z11
  let z16 := m6 + m22
  let z17 := z16 - z9
  let z18 := m8 - z3
  let z19 := m9 - z5
  let z20 := z19 - z7
  let z21 := m10 + z9
  let z22 := m11 + z11
  let z23 := m12 - z7
  let z24 := m16 - z2
  let z25 := z24 + m7
  -- output basis change (UNCOUNTED): the inverse transform,
  -- rows of the unimodular output basis
  let c11 := z21 - z23
  let c12 := z13
  let c13 := z25
  let c21 := z22
  let c22 := -z15
  let c23 := z18
  let c31 := z23
  let c32 := -z13 - z17
  let c33 := z18 + z20
  !![c11, c12, c13;
     c21, c22, c23;
     c31, c32, c33]

-- `rank23_add51_scheme` unfolds to a ~120-binding straight-line
-- program; a single elaboration exceeds the default heartbeat budget.
set_option maxHeartbeats 1600000 in
-- Plain `simp` evaluates the concrete `!![..] i j` indexing via
-- Mathlib's matrix simprocs; the explicit lemma set is version-fragile,
-- so the flexible-`simp` lint is accepted here (as in Mm/Matrix.lean).
set_option linter.flexible false in
/-- The scheme computes the matrix product, in Mathlib's own terms. -/
theorem rank23_add51_scheme_eq_mul (A B : Matrix (Fin 3) (Fin 3) R) :
    rank23_add51_scheme A B = A * B := by
  simp only [rank23_add51_scheme]
  ext i j
  fin_cases i <;> fin_cases j <;>
    simp [Matrix.mul_apply, Fin.sum_univ_three] <;>
    noncomm_ring
